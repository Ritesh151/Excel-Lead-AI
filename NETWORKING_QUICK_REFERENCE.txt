Error: SocketTimeoutException
  Cause: Backend offline or very slow network
  Fix: Check npm start, check firewall

Error: ConnectException: Connection refused
  Cause: Port 3000 not open on backend
  Fix: sudo ufw allow 3000, restart backend

Error: UnknownHostException
  Cause: Wrong backend IP address
  Fix: Check BACKEND_HOST in local.properties

Error: WebSocket FAILED state
  Cause: Backend unreachable or crashed
  Fix: Run network diagnostics, check backend logs

Error: Campaign Start Failed
  Cause: Network issue or backend error
  Fix: Check diagnostics output, verify backend health

// Create networking stack
val networking = NetworkingInitializer(context)

// Initialize WebSocket
networking.initializeWebSocket(
  onStateChange = { state -> updateUI(state) },
  onMessage = { event, payload -> handleEvent(event) },
  onError = { error -> showError(error) }
)

// Before starting campaign
fun startAutomation() {
  networking.runDiagnostics(viewModel) { report ->
    if (!report.isHealthy) {
      showDialog(report.recommendations)
      return
    }
    networking.startCampaign("Campaign") { id, leads ->
      // Campaign started
    }
  }
}

// Cleanup
override fun onCleared() {
  super.onCleared()
  networking.shutdown()
}
